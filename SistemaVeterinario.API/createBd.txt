-- ========================================================

-- 1. Criação das Tabelas
CREATE TABLE TB_TUTOR (
    ID_TUTOR NUMBER(10) PRIMARY KEY,
    NOME_TUTOR VARCHAR2(100) NOT NULL,
    CPF_TUTOR VARCHAR2(14) NOT NULL UNIQUE
);

CREATE TABLE TB_VETERINARIO (
    ID_VETERINARIO NUMBER(10) PRIMARY KEY,
    NOME_VETERINARIO VARCHAR2(100) NOT NULL,
    CRM_VETERINARIO VARCHAR2(20) NOT NULL UNIQUE,
    ESPECIALIDADE_VETERINARIO VARCHAR2(50) NOT NULL,
    TELEFONE_VETERINARIO VARCHAR2(20),
    EMAIL_VETERINARIO VARCHAR2(100),
    STATUS_VETERINARIO VARCHAR2(20),
    DATA_CADASTRO_VETERINARIO DATE DEFAULT SYSDATE NOT NULL
);

CREATE TABLE TB_PET (
    ID_PET NUMBER(10) PRIMARY KEY,
    ID_TUTOR NUMBER(10) NOT NULL,
    NOME_PET VARCHAR2(100) NOT NULL,
    ESPECIE_PET VARCHAR2(50) NOT NULL,
    RACA_PET VARCHAR2(80) NOT NULL,
    SEXO_PET VARCHAR2(10) NOT NULL,
    DATA_NASCIMENTO_PET DATE NOT NULL,
    PESO_PET NUMBER(10, 2) NOT NULL,
    COR_PET VARCHAR2(50) NOT NULL,
    DATA_CADASTRO_PET DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT FK_PET_TUTOR FOREIGN KEY (ID_TUTOR) REFERENCES TB_TUTOR(ID_TUTOR)
);

-- 2. Sequences e Triggers para Auto-Incremento (O segredo do banco perfeito)
-- Sequência para Tutor
CREATE SEQUENCE SEQ_TUTOR START WITH 1 INCREMENT BY 1;
CREATE OR REPLACE TRIGGER TRG_TUTOR_ID
BEFORE INSERT ON TB_TUTOR FOR EACH ROW
BEGIN
  IF :NEW.ID_TUTOR IS NULL THEN
    SELECT SEQ_TUTOR.NEXTVAL INTO :NEW.ID_TUTOR FROM DUAL;
  END IF;
END;
/

-- Sequência para Veterinário
CREATE SEQUENCE SEQ_VETERINARIO START WITH 1 INCREMENT BY 1;
CREATE OR REPLACE TRIGGER TRG_VETERINARIO_ID
BEFORE INSERT ON TB_VETERINARIO FOR EACH ROW
BEGIN
  IF :NEW.ID_VETERINARIO IS NULL THEN
    SELECT SEQ_VETERINARIO.NEXTVAL INTO :NEW.ID_VETERINARIO FROM DUAL;
  END IF;
END;
/

-- Sequência para Pet
CREATE SEQUENCE SEQ_PET START WITH 1 INCREMENT BY 1;
CREATE OR REPLACE TRIGGER TRG_PET_ID
BEFORE INSERT ON TB_PET FOR EACH ROW
BEGIN
  IF :NEW.ID_PET IS NULL THEN
    SELECT SEQ_PET.NEXTVAL INTO :NEW.ID_PET FROM DUAL;
  END IF;
END;
/
