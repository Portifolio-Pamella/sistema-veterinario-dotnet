-- 1. LIMPEZA TOTAL (Remoção segura)
DROP TRIGGER TRG_VET_ID;
DROP TRIGGER TRG_TUTOR_ID;
DROP SEQUENCE SEQ_VET;
DROP SEQUENCE SEQ_TUTOR;
DROP TABLE TB_PET CASCADE CONSTRAINTS;
DROP TABLE TB_TUTOR CASCADE CONSTRAINTS;
DROP TABLE TB_VETERINARIO CASCADE CONSTRAINTS;

-- 2. CRIAÇÃO DAS TABELAS
CREATE TABLE TB_VETERINARIO (
    ID_VETERINARIO NUMBER(10) PRIMARY KEY,
    NOME_VETERINARIO VARCHAR2(100) NOT NULL,
    CRM_VETERINARIO VARCHAR2(20) NOT NULL UNIQUE,
    ESPECIALIDADE_VETERINARIO VARCHAR2(50) NOT NULL,
    TELEFONE_VETERINARIO VARCHAR2(20),
    EMAIL_VETERINARIO VARCHAR2(100),
    STATUS_VETERINARIO VARCHAR2(20),
    DATA_CADASTRO_VETERINARIO DATE DEFAULT SYSDATE
);

CREATE TABLE TB_TUTOR (
    ID_TUTOR NUMBER(10) PRIMARY KEY,
    NOME_TUTOR VARCHAR2(100) NOT NULL,
    CPF_TUTOR VARCHAR2(14) NOT NULL UNIQUE,
    TELEFONE_TUTOR VARCHAR2(20) NOT NULL,
    EMAIL_TUTOR VARCHAR2(100) NOT NULL UNIQUE,
    CEP_TUTOR VARCHAR2(10) NOT NULL,
    RUA_TUTOR VARCHAR2(100) NOT NULL,
    NUMERO_TUTOR VARCHAR2(10) NOT NULL,
    COMPLEMENTO_TUTOR VARCHAR2(50),
    BAIRRO_TUTOR VARCHAR2(50) NOT NULL,
    CIDADE_TUTOR VARCHAR2(50) NOT NULL,
    ESTADO_TUTOR VARCHAR2(2) NOT NULL,
    DATA_CADASTRO_TUTOR DATE DEFAULT SYSDATE NOT NULL
);

CREATE TABLE TB_PET (
    ID_PET NUMBER(10) PRIMARY KEY,
    ID_TUTOR NUMBER(10) NOT NULL,
    NOME_PET VARCHAR2(100) NOT NULL,
    ESPECIE_PET VARCHAR2(50) NOT NULL,
    RACA_PET VARCHAR2(50) NOT NULL,
    SEXO_PET VARCHAR2(10) NOT NULL,
    DATA_NASCIMENTO_PET DATE NOT NULL,
    PESO_PET NUMBER(5,2) NOT NULL,
    COR_PET VARCHAR2(30) NOT NULL,
    DATA_CADASTRO_PET DATE DEFAULT SYSDATE NOT NULL,
    CONSTRAINT FK_PET_TUTOR FOREIGN KEY (ID_TUTOR) REFERENCES TB_TUTOR(ID_TUTOR)
);

-- 3. SEQUENCES
CREATE SEQUENCE SEQ_TUTOR START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE SEQ_VET START WITH 1 INCREMENT BY 1;

-- 4. TRIGGERS PARA IDs AUTOMÁTICOS
CREATE OR REPLACE TRIGGER TRG_TUTOR_ID
BEFORE INSERT ON TB_TUTOR
FOR EACH ROW
BEGIN
    IF :NEW.ID_TUTOR IS NULL THEN
        SELECT SEQ_TUTOR.NEXTVAL INTO :NEW.ID_TUTOR FROM DUAL;
    END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_VET_ID
BEFORE INSERT ON TB_VETERINARIO
FOR EACH ROW
BEGIN
    IF :NEW.ID_VETERINARIO IS NULL THEN
        SELECT SEQ_VET.NEXTVAL INTO :NEW.ID_VETERINARIO FROM DUAL;
    END IF;
END;
/
